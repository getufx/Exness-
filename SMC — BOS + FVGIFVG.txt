# indie:lang_version = 5
# =============================================================================
# SMC — BOS + FVG / IFVG  (Smart Money Concepts indicator)
# =============================================================================
# f
# OVERVIEW
# --------
# This indicator visualises four core Smart Money Concepts (SMC) structures on
# the price chart, all drawn as Rectangle zones or LineSegment lines:
#
#   Break of Structure (BOS)
#   ~~~~~~~~~~~~~~~~~~~~~~~~
#   A BOS occurs when price closes BEYOND the most-recent confirmed swing high
#   (bullish BOS — green dotted line) or swing low (bearish BOS — red dotted
#   line).  In SMC, a BOS signals that institutional order-flow has shifted:
#   a bullish BOS suggests buyers are in control and the structure is now
#   trending up; a bearish BOS suggests sellers are in control.
#   The script also draws a forward-looking "pending BOS" dotted line extending
#   from the last unbroken pivot to the current bar, so the trader can see
#   WHICH level must be broken next to produce a new BOS.
#
#   Fair Value Gap (FVG)
#   ~~~~~~~~~~~~~~~~~~~~
#   An FVG is a 3-bar price imbalance where one side of the market was not
#   fully traded.  A bullish FVG forms when bar[-1].low > bar[-3].high (a gap
#   UP between candle 3 and candle 1, anchored at the middle candle).  A
#   bearish FVG is the mirror: bar[-1].high < bar[-3].low.  In SMC, FVGs are
#   regarded as areas of unfinished business that price tends to revisit
#   ("fill" or "mitigate") before continuing the prevailing trend.
#   Active FVGs are drawn as opaque green (bullish) or red (bearish) rectangles.
#
#   Inverse Fair Value Gap (IFVG)
#   ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
#   When price returns into an FVG and closes inside or through it, the FVG is
#   said to be "mitigated."  At that point the FVG is PROMOTED into an Inverse
#   FVG (IFVG).  In SMC theory, a mitigated bullish FVG can now act as
#   resistance (flipped), and a mitigated bearish FVG as support.  IFVGs are
#   drawn with the same price range but at a much lower opacity (alpha 0.1) to
#   distinguish them visually from active FVGs.
#
#   Failed BOS (FBOS)
#   ~~~~~~~~~~~~~~~~~
#   A Failed BOS occurs when price WICKS beyond a pivot high/low (suggesting
#   a BOS) but the candle CLOSES back on the originating side — i.e., the
#   breakout failed.  In SMC this is an important liquidity sweep / stop-hunt
#   signal: smart money engineered the breakout to collect stop orders, then
#   reversed.  FBOS zones are centred on the swept pivot and sized using ATR,
#   drawn as semi-transparent golden (bullish FBOS) or purple (bearish FBOS)
#   rectangles.
#
# DRAWING MODEL
# -------------
# Indie requires drawing objects to be pre-allocated and reused across bars
# (the platform caps per-bar drawing changes at ~100). Therefore all Rectangle
# slots are created once in __init__ with NaN coordinates and then repositioned
# every bar via a cache-diffing helper (_draw_rect_cached) that skips redundant
# chart.draw calls.  BOS confirmed-break lines are accumulated in a capped list
# (max 10); the "pending BOS" line is a single reused LineSegment per direction.
# =============================================================================

from math import nan, isnan
from indie import indicator, MainContext, param, color, Color, line_style, plot, Var
from indie.algorithms import PivotHighLow, Atr
from indie.drawings import LineSegment, Rectangle, AbsolutePosition, LabelAbs, callout_position

# Maximum number of FVG rectangle slots pre-allocated in __init__.
# This is the hard upper bound that governs the size of the drawing pool and
# all corresponding data lists.  The user-facing "Max FVGs" parameter is
# bounded to [1, MAX_FVG_CAPACITY], so the pool is always large enough.
MAX_FVG_CAPACITY = 10

# Maximum number of IFVG rectangle slots pre-allocated in __init__.
# IFVGs are rarer than FVGs (they are promoted FROM FVGs after mitigation),
# so a smaller pool of 7 is sufficient while keeping memory use low.
MAX_IFVG_CAPACITY = 7

# Maximum number of Failed BOS rectangle slots pre-allocated in __init__.
# Sized the same as FVGs (10) because FBOS events can occur as frequently as
# FVGs on volatile markets.
MAX_FBOS_CAPACITY = 10

# _feq — float equality with NaN-awareness
# -----------------------------------------
# Purpose : Compare two floats safely when either may be NaN.
#           Python's IEEE 754 rule is NaN != NaN, which would cause every
#           cache comparison to report a change when both values are NaN,
#           triggering spurious redundant chart.draw calls.
# Params  : a, b — the two floats to compare.
# Returns : True if both are NaN (treated as equal) or if a == b exactly.
def _feq(a: float, b: float) -> bool:
    if isnan(a) and isnan(b):
        return True
    return a == b

# _rect_changed — detect whether a rectangle's coordinates have changed
# ----------------------------------------------------------------------
# Purpose : Before issuing a chart.draw call the script checks whether the
#           rectangle's four coordinate values (point_a time, point_a price,
#           point_b time, point_b price) differ from the values stored in the
#           4-element cache list.  If nothing changed, the draw is skipped,
#           which is the primary mechanism keeping per-bar drawing changes
#           well under the ~100 platform limit.
# Params  : cache — the 4-float list [pat, pap, pbt, pbp] from the previous draw.
#           pat   — point_a time (Unix seconds).
#           pap   — point_a price.
#           pbt   — point_b time (Unix seconds).
#           pbp   — point_b price.
# Returns : True if any coordinate has changed (draw needed); False otherwise.
def _rect_changed(cache: list[float], pat: float, pap: float, pbt: float, pbp: float) -> bool:
	if not _feq(cache[0], pat):
		return True
	if not _feq(cache[1], pap):
		return True
	if not _feq(cache[2], pbt):
		return True
	if not _feq(cache[3], pbp):
		return True
	return False

# _make_cache — build a list of per-slot coordinate caches
# ---------------------------------------------------------
# Purpose : Each drawing slot needs its own 4-float cache so that
#           _rect_changed can track the last-drawn coordinates independently
#           for every rectangle.  This function creates `capacity` such cache
#           entries pre-filled with [0.0, nan, 0.0, nan].  The time values are
#           initialised to 0.0 (valid timestamp sentinel) and the price values
#           to nan so that the very first draw on each slot is always triggered
#           (since real prices are never NaN after warmup).
# Params  : capacity — number of cache entries to create (matches the
#           corresponding MAX_*_CAPACITY constant).
# Returns : A list[list[float]] of length `capacity`.
def _make_cache(capacity: int) -> list[list[float]]:
	result: list[list[float]] = []
	i = 0
	while i < capacity:
		entry: list[float] = [0.0, nan, 0.0, nan]
		result.append(entry)
		i = i + 1
	return result
@indicator('SMC — BOS + FVG/IFVG', overlay_main_pane=True)

		)
		self._bos_bull_cache: list[float] = [0.0, nan, 0.0, nan]
		self._bos_bear_cache: list[float] = [0.0, nan, 0.0, nan]
		self._bull_fvg_t: list[float] = []
		self._bull_fvg_top: list[float] = []
		self._bull_fvg_bot: list[float] = []
		self._bull_fvg_active: list[bool] = []
		self._bull_fvg_end_t: list[float] = []
		self._bear_fvg_t: list[float] = []
		self._bear_fvg_top: list[float] = []
		self._bear_fvg_bot: list[float] = []
		self._bear_fvg_active: list[bool] = []
		self._bear_fvg_end_t: list[float] = []
		self._bull_ifvg_t: list[float] = []
		self._bull_ifvg_top: list[float] = []
		self._bull_ifvg_bot: list[float] = []
		self._bull_ifvg_active: list[bool] = []
		self._bull_ifvg_end_t: list[float] = []
		self._bear_ifvg_t: list[float] = []
		self._bear_ifvg_top: list[float] = []
		self._bear_ifvg_bot: list[float] = []

		self._bear_ifvg_end_t: list[float] = []
		self._bull_fvg_rects: list[Rectangle] = []
		self._bear_fvg_rects: list[Rectangle] = []
		self._bull_ifvg_rects: list[Rectangle] = []
		self._bear_ifvg_rects: list[Rectangle] = []
		bull_fvg_bg = color.rgba(82, 250, 141, 1)
		bear_fvg_bg = color.rgba(250, 82, 82, 1)
		bull_ifvg_bg = color.rgba(82, 250, 141, 0.1)
		bear_ifvg_bg = color.rgba(250, 82, 82, 0.1)
		no_border = color.rgba(0, 0, 0, 0.0)
		i = 0
		while i < MAX_FVG_CAPACITY:
			self._bull_fvg_rects.append(Rectangle(
				AbsolutePosition(0.0, nan), AbsolutePosition(1.0, nan),
				line_color=no_border, bg_color=bull_fvg_bg,

			self._bear_fvg_rects.append(Rectangle(
				AbsolutePosition(0.0, nan), AbsolutePosition(1.0, nan),
				line_color=no_border, bg_color=bear_fvg_bg,
				line_width=0, middle_line_width=0, middle_line_color=no_border,
			))
			i = i + 1
		i = 0
		while i < MAX_IFVG_CAPACITY:
			self._bull_ifvg_rects.append(Rectangle(
				AbsolutePosition(0.0, nan), AbsolutePosition(1.0, nan),
				line_color=no_border, bg_color=bull_ifvg_bg,
				line_width=0, middle_line_width=0, middle_line_color=no_border,

			self._bear_ifvg_rects.append(Rectangle(

			i = i + 1
		self._last_fvg_bar: Var[int] = self.new_var(-1)
		self._bull_fvg_cache: list[list[float]] = _make_cache(MAX_FVG_CAPACITY)
		self._bear_fvg_cache: list[list[float]] = _make_cache(MAX_FVG_CAPACITY)
		self._bull_ifvg_cache: list[list[float]] = _make_cache(MAX_IFVG_CAPACITY)
		self._bear_ifvg_cache: list[list[float]] = _make_cache(MAX_IFVG_CAPACITY)
		self._last_failed_bos_bull_price: float = nan
		self._last_failed_bos_bear_price: float = nan
		self._bull_fbos_t: list[float] = []
		self._bull_fbos_top: list[float] = []

		self._bull_fbos_active: list[bool] = []
		self._bull_fbos_end_t: list[float] = []
		self._bear_fbos_t: list[float] = []

		self._bull_fbos_rects: list[Rectangle] = []

			self._bear_fbos_rects.append(Rectangle(

				line_width=0, middle_line_width=0, middle_line_color=no_border,
			))
			i = i + 1
		self._bull_fbos_cache: list[list[float]] = _make_cache(MAX_FBOS_CAPACITY)
		self._bear_fbos_cache: list[list[float]] = _make_cache(MAX_FBOS_CAPACITY)
		fib_color = color.rgba(255, 165, 0, 1)
		self._fib1_line: LineSegment = LineSegment(

			color=fib_color, line_style=line_style.DOTTED, line_width=1,
		)
		self._fib1_cache: list[float] = [0.0, nan, 0.0, nan]
		self._fib2_cache: list[float] = [0.0, nan, 0.0, nan]
		self._fib1_label: LabelAbs = LabelAbs(
			'', no_pos,
			text_color=fib_color, callout_position=callout_position.TOP_LEFT,

			'', no_pos,
			text_color=fib_color, callout_position=callout_position.TOP_LEFT,
			bg_color=color.TRANSPARENT, font_size=11,
		)
	def _draw_bos_line(self, swing_time: float, swing_price: float, now_time: float,
			line_color: Color, lines_lst: Var[list[LineSegment]]) -> None:
		seg = LineSegment(
			AbsolutePosition(swing_time, swing_price),
			AbsolutePosition(now_time, swing_price),
			color=line_color,
			line_style=line_style.DOTTED,
			line_width=2,
		)
		self.chart.draw(seg)
		lst = lines_lst.get()
		if len(lst) >= 10:
			self.chart.erase(lst[0])
			del lst[0]
		lst.append(seg)
	def _add_gap(self, t_list: list[float], top_list: list[float], bot_list: list[float],
			active_list: list[bool], end_list: list[float],
			t: float, top: float, bot: float, max_len: int) -> None:
		t_list.append(t)
		top_list.append(top)
		bot_list.append(bot)
		active_list.append(True)
		end_list.append(nan)
		if len(t_list) > max_len:
			remove_idx = 0
			i = 0
			while i < len(active_list) - 1:
				if not active_list[i]:
					remove_idx = i
					break
				i = i + 1
			t_list.pop(remove_idx)
			top_list.pop(remove_idx)
			bot_list.pop(remove_idx)
			active_list.pop(remove_idx)
			end_list.pop(remove_idx)
	def _draw_rect_cached(self, rect: Rectangle, cache: list[float],
			pat: float, pap: float, pbt: float, pbp: float) -> None:
		if _rect_changed(cache, pat, pap, pbt, pbp):
			rect.point_a = AbsolutePosition(pat, pap)
			rect.point_b = AbsolutePosition(pbt, pbp)
			self.chart.draw(rect)
			cache[0] = pat
			cache[1] = pap
			cache[2] = pbt
			cache[3] = pbp
	def calc(self, length: int, max_fvgs: int, show_fvgs: bool, show_ifvgs: bool, max_ifvgs: int, mode: str, max_failed_bos: int, fbos_filter_atr: float,
			show_fib_levels: bool, fib_level_1: float, fib_level_2: float):
		bar_idx = self.bar_index
		t0 = self.time[0]
		bar_duration_s = self.time_frame.to_minutes() * 60
		atr_series = Atr.new(14, 'RMA')
		atr_val = atr_series[0]
		half_height: float = nan
		if not isnan(atr_val) and atr_val > 0.0:
			half_height = atr_val * 0.5
		if bar_idx == 0:
			i = 0
			while i < MAX_FVG_CAPACITY:
				self._bull_fvg_rects[i].point_a = AbsolutePosition(t0, nan)
				self._bull_fvg_rects[i].point_b = AbsolutePosition(t0, nan)
				self.chart.draw(self._bull_fvg_rects[i])
				self._bear_fvg_rects[i].point_a = AbsolutePosition(t0, nan)
				self._bear_fvg_rects[i].point_b = AbsolutePosition(t0, nan)
				self.chart.draw(self._bear_fvg_rects[i])
				i = i + 1
			i = 0
			while i < MAX_IFVG_CAPACITY:
				self._bull_ifvg_rects[i].point_a = AbsolutePosition(t0, nan)
				self._bull_ifvg_rects[i].point_b = AbsolutePosition(t0, nan)
				self.chart.draw(self._bull_ifvg_rects[i])
				self._bear_ifvg_rects[i].point_a = AbsolutePosition(t0, nan)
				self._bear_ifvg_rects[i].point_b = AbsolutePosition(t0, nan)
				self.chart.draw(self._bear_ifvg_rects[i])
				i = i + 1
			i = 0
			while i < MAX_FBOS_CAPACITY:
				self._bull_fbos_rects[i].point_a = AbsolutePosition(t0, nan)
				self._bull_fbos_rects[i].point_b = AbsolutePosition(t0, nan)
				self.chart.draw(self._bull_fbos_rects[i])
				self._bear_fbos_rects[i].point_a = AbsolutePosition(t0, nan)
				self._bear_fbos_rects[i].point_b = AbsolutePosition(t0, nan)
				self.chart.draw(self._bear_fbos_rects[i])
				i = i + 1
			self.chart.draw(self._next_bos_bull_seg)
			self.chart.draw(self._next_bos_bear_seg)
			self.chart.draw(self._fib1_line)
			self.chart.draw(self._fib2_line)
			self.chart.draw(self._fib1_label)
			self.chart.draw(self._fib2_label)
		if self.is_closed_bar:
			ph, _ = PivotHighLow.new(self.high, left_bars=length, right_bars=length)
			_, pl = PivotHighLow.new(self.low, left_bars=length, right_bars=length)
			now_time = self.time[0]
			cur_close = self.close[0]
			if not isnan(ph[0]):
				self._last_ph_price = ph[0]
				self._last_ph_time = self.time[length]
			if not isnan(pl[0]):
				self._last_pl_price = pl[0]
				self._last_pl_time = self.time[length]
			if not isnan(self._last_ph_price):
				if cur_close > self._last_ph_price and self._last_ph_price != self._last_bos_bull_price:
					self._draw_bos_line(self._last_ph_time, self._last_ph_price, now_time, color.GREEN, self._lines_bos_bull)
					self._last_bos_bull_price = self._last_ph_price
					self._last_bos_direction = 'bull'
			if not isnan(self._last_pl_price):
				if cur_close < self._last_pl_price and self._last_pl_price != self._last_bos_bear_price:
					self._draw_bos_line(self._last_pl_time, self._last_pl_price, now_time, color.RED, self._lines_bos_bear)
					self._last_bos_bear_price = self._last_pl_price
					self._last_bos_direction = 'bear'
			if not isnan(self._last_ph_price):
				if self.high[0] > self._last_ph_price and cur_close <= self._last_ph_price and (fbos_filter_atr == 0.0 or (not isnan(atr_val) and atr_val > 0.0 and (self.high[0] - self._last_ph_price) >= fbos_filter_atr * atr_val)):
					if self._last_ph_price != self._last_failed_bos_bull_price:
						swing_p = self._last_ph_price
						zone_top: float = swing_p + swing_p * 0.001
						zone_bot: float = swing_p - swing_p * 0.001
						if not isnan(half_height):
							zone_top = swing_p + half_height
							zone_bot = swing_p - half_height
						self._add_gap(
							self._bull_fbos_t, self._bull_fbos_top, self._bull_fbos_bot,
							self._bull_fbos_active, self._bull_fbos_end_t,
							self._last_ph_time, zone_top, zone_bot, max_failed_bos,
						)
						self._last_failed_bos_bull_price = swing_p
			if not isnan(self._last_pl_price):
				if self.low[0] < self._last_pl_price and cur_close >= self._last_pl_price and (fbos_filter_atr == 0.0 or (not isnan(atr_val) and atr_val > 0.0 and (self._last_pl_price - self.low[0]) >= fbos_filter_atr * atr_val)):
					if self._last_pl_price != self._last_failed_bos_bear_price:
						swing_p = self._last_pl_price
						b_zone_top: float = swing_p + swing_p * 0.001
						b_zone_bot: float = swing_p - swing_p * 0.001
						if not isnan(half_height):
							b_zone_top = swing_p + half_height
							b_zone_bot = swing_p - half_height
						self._add_gap(
							self._bear_fbos_t, self._bear_fbos_top, self._bear_fbos_bot,
							self._bear_fbos_active, self._bear_fbos_end_t,
							self._last_pl_time, b_zone_top, b_zone_bot, max_failed_bos,
						)
						self._last_failed_bos_bear_price = swing_p
			i = 0
			while i < len(self._bull_fbos_active):
				if self._bull_fbos_active[i]:
					if cur_close > self._bull_fbos_top[i]:
						self._bull_fbos_active[i] = False
						self._bull_fbos_end_t[i] = t0
				i = i + 1
			i = 0
			while i < len(self._bear_fbos_active):
				if self._bear_fbos_active[i]:
					if cur_close < self._bear_fbos_bot[i]:
						self._bear_fbos_active[i] = False
						self._bear_fbos_end_t[i] = t0
				i = i + 1
			# --- Fib levels between the most recent swing high and swing low ---
			# Direction follows the last confirmed BOS: after a bullish BOS,
			# levels are measured as a pullback DOWN from the swing high
			# (price = high - range * level); after a bearish BOS, measured
			# as a pullback UP from the swing low (price = low + range * level).
			# Recomputes every time the swing pair or the BOS direction
			# changes, so it always reflects the latest known state.
			if show_fib_levels and self._last_bos_direction != '' and not isnan(self._last_ph_price) and not isnan(self._last_pl_price):
				swing_high = self._last_ph_price
				swing_low = self._last_pl_price
				swing_range = swing_high - swing_low
				if swing_range > 0.0:
					fib_color: Color = color.GREEN
					dir_label = 'Bull '
					fib1_price: float = nan
					fib2_price: float = nan
					if self._last_bos_direction == 'bull':
						fib1_price = swing_high - swing_range * fib_level_1
						fib2_price = swing_high - swing_range * fib_level_2
					else:
						fib_color = color.RED
						dir_label = 'Bear '
						fib1_price = swing_low + swing_range * fib_level_1
						fib2_price = swing_low + swing_range * fib_level_2
					anchor_time = self._last_ph_time
					if self._last_pl_time > anchor_time:
						anchor_time = self._last_pl_time
					if _rect_changed(self._fib1_cache, anchor_time, fib1_price, t0, fib1_price):
						self._fib1_line.point_a = AbsolutePosition(anchor_time, fib1_price)
						self._fib1_line.point_b = AbsolutePosition(t0, fib1_price)
						self._fib1_line.color = fib_color
						self.chart.draw(self._fib1_line)
						self._fib1_cache[0] = anchor_time
						self._fib1_cache[1] = fib1_price
						self._fib1_cache[2] = t0
						self._fib1_cache[3] = fib1_price
						self._fib1_label.position = AbsolutePosition(t0, fib1_price)
						self._fib1_label.text_color = fib_color
						self._fib1_label.text = dir_label + str(round(fib_level_1 * 100, 1)) + '% (' + str(round(fib1_price, self.info.price_precision)) + ')'
						self.chart.draw(self._fib1_label)
					if _rect_changed(self._fib2_cache, anchor_time, fib2_price, t0, fib2_price):
						self._fib2_line.point_a = AbsolutePosition(anchor_time, fib2_price)
						self._fib2_line.point_b = AbsolutePosition(t0, fib2_price)
						self._fib2_line.color = fib_color
						self.chart.draw(self._fib2_line)
						self._fib2_cache[0] = anchor_time
						self._fib2_cache[1] = fib2_price
						self._fib2_cache[2] = t0
						self._fib2_cache[3] = fib2_price
						self._fib2_label.position = AbsolutePosition(t0, fib2_price)
						self._fib2_label.text_color = fib_color
						self._fib2_label.text = dir_label + str(round(fib_level_2 * 100, 1)) + '% (' + str(round(fib2_price, self.info.price_precision)) + ')'
						self.chart.draw(self._fib2_label)
			close0 = self.close[0]
			if bar_idx >= 4 and bar_idx != self._last_fvg_bar.get():
				self._last_fvg_bar.set(bar_idx)
				if not isnan(self.low[1]) and not isnan(self.high[3]):
					if self.low[1] > self.high[3]:
						self._add_gap(self._bull_fvg_t, self._bull_fvg_top, self._bull_fvg_bot,
							self._bull_fvg_active, self._bull_fvg_end_t,
							self.time[2], self.low[1], self.high[3], max_fvgs)
				if not isnan(self.high[1]) and not isnan(self.low[3]):
					if self.high[1] < self.low[3]:
						self._add_gap(self._bear_fvg_t, self._bear_fvg_top, self._bear_fvg_bot,
							self._bear_fvg_active, self._bear_fvg_end_t,
							self.time[2], self.low[3], self.high[1], max_fvgs)
			i = 0
			while i < len(self._bull_fvg_active):
				if self._bull_fvg_active[i]:
					mitigated = False
					if close0 >= self._bull_fvg_bot[i] and close0 <= self._bull_fvg_top[i]:
						mitigated = True
					elif close0 < self._bull_fvg_bot[i]:
						mitigated = True
					if mitigated:
						self._bull_fvg_active[i] = False
						self._bull_fvg_end_t[i] = t0
						self._add_gap(self._bull_ifvg_t, self._bull_ifvg_top, self._bull_ifvg_bot,
							self._bull_ifvg_active, self._bull_ifvg_end_t,
							self._bull_fvg_t[i], self._bull_fvg_top[i], self._bull_fvg_bot[i], max_ifvgs)
				i = i + 1
			i = 0
			while i < len(self._bear_fvg_active):
				if self._bear_fvg_active[i]:
					mitigated = False
					if close0 >= self._bear_fvg_bot[i] and close0 <= self._bear_fvg_top[i]:
						mitigated = True
					elif close0 > self._bear_fvg_top[i]:
						mitigated = True
					if mitigated:
						self._bear_fvg_active[i] = False
						self._bear_fvg_end_t[i] = t0
						self._add_gap(self._bear_ifvg_t, self._bear_ifvg_top, self._bear_ifvg_bot,
							self._bear_ifvg_active, self._bear_ifvg_end_t,
							self._bear_fvg_t[i], self._bear_fvg_top[i], self._bear_fvg_bot[i], max_ifvgs)
				i = i + 1
			i = 0
			while i < len(self._bull_ifvg_active):
				if self._bull_ifvg_active[i]:
					if close0 > self._bull_ifvg_top[i]:
						self._bull_ifvg_active[i] = False
						self._bull_ifvg_end_t[i] = t0
				i = i + 1
			i = 0
			while i < len(self._bear_ifvg_active):
				if self._bear_ifvg_active[i]:
					if close0 < self._bear_ifvg_bot[i]:
						self._bear_ifvg_active[i] = False
						self._bear_ifvg_end_t[i] = t0
				i = i + 1
		i = 0
		while i < MAX_FVG_CAPACITY:
			if show_fvgs and i < len(self._bull_fvg_t) and (mode == 'Historical' or self._bull_fvg_active[i]):
				end_time = t0 + bar_duration_s if self._bull_fvg_active[i] else self._bull_fvg_end_t[i] + bar_duration_s
				self._draw_rect_cached(self._bull_fvg_rects[i], self._bull_fvg_cache[i],
					self._bull_fvg_t[i], self._bull_fvg_top[i], end_time, self._bull_fvg_bot[i])
			else:
				self._draw_rect_cached(self._bull_fvg_rects[i], self._bull_fvg_cache[i],
					t0, nan, t0, nan)
			i = i + 1
		i = 0
		while i < MAX_FVG_CAPACITY:
			if show_fvgs and i < len(self._bear_fvg_t) and (mode == 'Historical' or self._bear_fvg_active[i]):
				end_time = t0 + bar_duration_s if self._bear_fvg_active[i] else self._bear_fvg_end_t[i] + bar_duration_s
				self._draw_rect_cached(self._bear_fvg_rects[i], self._bear_fvg_cache[i],
					self._bear_fvg_t[i], self._bear_fvg_top[i], end_time, self._bear_fvg_bot[i])
			else:
				self._draw_rect_cached(self._bear_fvg_rects[i], self._bear_fvg_cache[i],
					t0, nan, t0, nan)
			i = i + 1
		i = 0
		while i < MAX_IFVG_CAPACITY:
			if show_ifvgs and i < len(self._bull_ifvg_t) and (mode == 'Historical' or self._bull_ifvg_active[i]):
				end_time = t0 + bar_duration_s if self._bull_ifvg_active[i] else self._bull_ifvg_end_t[i] + bar_duration_s
				self._draw_rect_cached(self._bull_ifvg_rects[i], self._bull_ifvg_cache[i],
					self._bull_ifvg_t[i], self._bull_ifvg_top[i], end_time, self._bull_ifvg_bot[i])
			else:
				self._draw_rect_cached(self._bull_ifvg_rects[i], self._bull_ifvg_cache[i],
					t0, nan, t0, nan)
			i = i + 1
		i = 0
		while i < MAX_IFVG_CAPACITY:
			if show_ifvgs and i < len(self._bear_ifvg_t) and (mode == 'Historical' or self._bear_ifvg_active[i]):
				end_time = t0 + bar_duration_s if self._bear_ifvg_active[i] else self._bear_ifvg_end_t[i] + bar_duration_s
				self._draw_rect_cached(self._bear_ifvg_rects[i], self._bear_ifvg_cache[i],
					self._bear_ifvg_t[i], self._bear_ifvg_top[i], end_time, self._bear_ifvg_bot[i])
			else:
				self._draw_rect_cached(self._bear_ifvg_rects[i], self._bear_ifvg_cache[i],
					t0, nan, t0, nan)
			i = i + 1
		i = 0
		while i < MAX_FBOS_CAPACITY:
			if i < len(self._bull_fbos_t) and (mode == 'Historical' or self._bull_fbos_active[i]):
				end_time = t0 + bar_duration_s if self._bull_fbos_active[i] else self._bull_fbos_end_t[i] + bar_duration_s
				self._draw_rect_cached(self._bull_fbos_rects[i], self._bull_fbos_cache[i],
					self._bull_fbos_t[i], self._bull_fbos_top[i], end_time, self._bull_fbos_bot[i])
			else:
				self._draw_rect_cached(self._bull_fbos_rects[i], self._bull_fbos_cache[i],
					t0, nan, t0, nan)
			i = i + 1
		i = 0
		while i < MAX_FBOS_CAPACITY:
			if i < len(self._bear_fbos_t) and (mode == 'Historical' or self._bear_fbos_active[i]):
				end_time = t0 + bar_duration_s if self._bear_fbos_active[i] else self._bear_fbos_end_t[i] + bar_duration_s
				self._draw_rect_cached(self._bear_fbos_rects[i], self._bear_fbos_cache[i],
					self._bear_fbos_t[i], self._bear_fbos_top[i], end_time, self._bear_fbos_bot[i])
			else:
				self._draw_rect_cached(self._bear_fbos_rects[i], self._bear_fbos_cache[i],
					t0, nan, t0, nan)
			i = i + 1
		has_bull = not isnan(self._last_ph_price) and self._last_ph_price != self._last_bos_bull_price
		has_bear = not isnan(self._last_pl_price) and self._last_pl_price != self._last_bos_bear_price
		bull_pat: float = t0
		bull_pap: float = nan
		bull_pbt: float = t0
		bull_pbp: float = nan
		if has_bull:
			bull_pat = self._last_ph_time
			bull_pap = self._last_ph_price
			bull_pbt = t0 + bar_duration_s
			bull_pbp = self._last_ph_price
		if _rect_changed(self._bos_bull_cache, bull_pat, bull_pap, bull_pbt, bull_pbp):
			self._next_bos_bull_seg.point_a = AbsolutePosition(bull_pat, bull_pap)
			self._next_bos_bull_seg.point_b = AbsolutePosition(bull_pbt, bull_pbp)
			self.chart.draw(self._next_bos_bull_seg)
			self._bos_bull_cache[0] = bull_pat
			self._bos_bull_cache[1] = bull_pap
			self._bos_bull_cache[2] = bull_pbt
			self._bos_bull_cache[3] = bull_pbp
		bear_pat: float = t0
		bear_pap: float = nan
		bear_pbt: float = t0
		bear_pbp: float = nan
		if has_bear:
			bear_pat = self._last_pl_time
			bear_pap = self._last_pl_price
			bear_pbt = t0 + bar_duration_s
			bear_pbp = self._last_pl_price
		if _rect_changed(self._bos_bear_cache, bear_pat, bear_pap, bear_pbt, bear_pbp):
			self._next_bos_bear_seg.point_a = AbsolutePosition(bear_pat, bear_pap)
			self._next_bos_bear_seg.point_b = AbsolutePosition(bear_pbt, bear_pbp)
			self.chart.draw(self._next_bos_bear_seg)
			self._bos_bear_cache[0] = bear_pat
			self._bos_bear_cache[1] = bear_pap
			self._bos_bear_cache[2] = bear_pbt
			self._bos_bear_cache[3] = bear_pbp
		return nan
